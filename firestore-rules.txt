rules_version = '2';

  match /databases/{database}/documents {

    // ===== CLIENTES =====
    match /clients/{document} {
      allow read, write: if true;
    }

    // ===== ORÇAMENTOS =====
    match /budgets/{document} {
      allow read, write: if true;
    }

    // ===== PRODUTOS =====
    match /products/{document} {
      allow read, write: if true;
    }

    // ===== SLIDES DO CARROSSEL =====
    match /hero_slides/{document} {
      allow read, write: if true;
    }

    // ===== BLOQUEIO PADRÃO =====
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
