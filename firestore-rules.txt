rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
        && request.auth.token.email != null
        && request.auth.token.email in ['admin@pcformatech.com'];
    }

    function isValidPresenceShape(sessionId) {
      return request.resource.data.keys().hasOnly(['sessionId', 'page', 'pagina', 'entrou', 'lastSeen', 'dispositivo'])
        && request.resource.data.sessionId == sessionId
        && request.resource.data.page is string
        && request.resource.data.page.size() <= 40
        && request.resource.data.pagina is string
        && request.resource.data.pagina.size() <= 120
        && request.resource.data.dispositivo in ['mobile', 'desktop']
        && request.resource.data.entrou is timestamp
        && request.resource.data.lastSeen is timestamp;
    }

    // ===== DADOS DO PAINEL ADMIN =====
    // clients, products, budgets, services ficam como documentos dentro de "data"
    match /data/{document} {
      allow read, write: if isAdmin();
    }

    // ===== IMAGENS DO CARROSSEL =====
    // O app acessa slider/images
    match /slider/{document} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ===== PRESENÇA / VISITANTES ONLINE =====
    // Cada visitante grava seu próprio documento de sessão
    match /presence/{sessionId} {
      allow read: if isAdmin();
      allow create, update: if isValidPresenceShape(sessionId);
      allow delete: if true;
    }

    // ===== BLOQUEIO PADRÃO =====
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
