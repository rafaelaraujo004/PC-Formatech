rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ===== DADOS DO PAINEL ADMIN =====
    // clients, products, budgets, services ficam como documentos dentro de "data"
    match /data/{document} {
      allow read, write: if true;
    }

    // ===== IMAGENS DO CARROSSEL =====
    // O app acessa slider/images
    match /slider/{document} {
      allow read, write: if true;
    }

    // ===== PRESENÇA / VISITANTES ONLINE =====
    // Cada visitante grava seu próprio documento de sessão
    match /presence/{sessionId} {
      allow read, write: if true;
    }

    // ===== BLOQUEIO PADRÃO =====
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
